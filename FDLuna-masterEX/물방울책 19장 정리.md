> raw 파일 읽기

- raw 파일은 그냥 바이트들이 연이어 저장된 파일일 뿐이다.
- 따라서 std::ifstream::read 호출 한 번으로 그 바이트들을 하나의 메모리 블록으로 읽을 수 있다.
- 다만 16비트 형식으로 읽을 때는 다르게 설정해야 한다.

```cpp
void CHill::LoadHeightMap(const tchar_t* pHightMapPath)
{
    vector<uint16_t> in(m_ivtxCntX * m_ivtxCntZ);
    ifstream inFile;
    inFile.open(pHightMapPath, ios_base::binary); // binary로 읽어야 함

    if (inFile)
    {
        // read(char* s, streamsize count);
        // s: 읽은 데이터를 채울 메모리 시작 주소
        // count: 읽을 바이트 수 -> 백터 길이 * 벡터 자료형의 크기

        inFile.read(
            (char*)&in[0], 
            streamsize(in.size() * sizeof(in[0]))
        );
        inFile.close();
    }

    m_vHeightmap.resize(m_ivtxCntX * m_ivtxCntZ);
    for (uint32_t i = 0; i < m_ivtxCntZ; ++i)
    {
        for (uint32_t j = 0; j < m_ivtxCntX; ++j)
        {
            uint32_t iSrcRow = (m_ivtxCntZ - 1) - i;        // 파일 자체가 z가 뒤집혀있기때문에 반전시켜줘야 한다.
                                                            // 뒤집기 -> 마지막 인덱스에서 빼기
            m_vHeightmap[i * m_ivtxCntX + j] = in[iSrcRow * m_ivtxCntX + j] / 65535.f * 600.f - 30.f; // 16비트로 나눠야함 2^16 = 65536
        }
    }
}
```

> f32_t CHill::GetHeight(f32_t x, f32_t z)

- 현재 x,z값이 높이 벡터의 어떤 사각형에 있는지 알아야 한다.
- 내림을 적용하여 사각형의 왼쪽 위 정점(A)을 구한 뒤 그 정점을 이용하여 나머지 정점 B,C,D를 구한다.
- 지형 내부에 있는지 확인한 뒤 높이 벡터의 인덱스로 사용할 수 있다.
- 왼쪽 위 꼭짓점과의 x,z 거리를 이용하여 윗 삼각형에 있는지 아래 삼각형에 있는지 알 수 있다.
- 이후 y값 반환.

```cpp
f32_t CHill::GetHeight(f32_t x, f32_t z)
{
    // 지형 생성시 사용한 x,z 값을 이용하여 row, col 값을 계산
    // float x = -m_fHalfWidth + j * m_fDx;
    // float z = m_fHalfDepth - i * m_fDz;

    f32_t col = (x + m_fHalfWidth) / m_fDx;
    f32_t row = (m_fHalfDepth - z) / m_fDz;

    // 내림 적용
    int32_t j = (int32_t)floorf(col);
    int32_t i = (int32_t)floorf(row);

    // 지형 내부에 있는지 확인
    if (i < 0 || j < 0 || i >= (int32_t)m_ivtxCntZ - 1 || j >= (int32_t)m_ivtxCntX - 1)
        return 0.f;

    // 사각형 네 꼭짓점 높이 구하기
    // 두르기 순서 ABC / CBD
    f32_t A = m_vHeightmap[i * m_ivtxCntX + j];
    f32_t B = m_vHeightmap[i * m_ivtxCntX + j + 1];
    f32_t C = m_vHeightmap[(i + 1) * m_ivtxCntX + j];
    f32_t D = m_vHeightmap[(i + 1) * m_ivtxCntX + j + 1];

    f32_t s = col - (f32_t)j;
    f32_t t = row - (f32_t)i;

    if (s + t <= 1.f) // 위쪽 삼각형
        return A + s * (B - A) + t * (C - A);
    else
        return D + (1.f - s) * (C - D) + (1.f - t) * (B - D);
}
```

> float4_t CHill::GetPlane(f32_t x, f32_t z)

- 그림자를 생성할 때 필요한 평면을 얻는 함수
- 높이를 구하는 GetHeight() 함수와 동일한 방식으로 진행 후 삼각형의 꼭짓점을 구한 뒤 평면을 구해서 반환한다.

```cpp
float4_t CHill::GetPlane(f32_t x, f32_t z)
{
    f32_t col = (x + m_fHalfWidth) / m_fDx;
    f32_t row = (m_fHalfDepth - z) / m_fDz;

    // 내림 적용
    int32_t j = (int32_t)floorf(col);
    int32_t i = (int32_t)floorf(row);

    // 지형 내부에 있는지 확인
    if (i < 0 || j < 0 || i >= (int32_t)m_ivtxCntZ - 1 || j >= (int32_t)m_ivtxCntX - 1)
        return float4_t{0.f,1.f,0.f,0.f};

    // 사각형 네 꼭짓점 높이 구하기
    // 두르기 순서 ABC / CBD
    f32_t A = m_vHeightmap[i * m_ivtxCntX + j];
    f32_t B = m_vHeightmap[i * m_ivtxCntX + j + 1];
    f32_t C = m_vHeightmap[(i + 1) * m_ivtxCntX + j];
    f32_t D = m_vHeightmap[(i + 1) * m_ivtxCntX + j + 1];

    f32_t s = col - (f32_t)j;
    f32_t t = row - (f32_t)i;

    if (s + t <= 1.f) // 위쪽 삼각형
    {
        vector_t p0 = vector_t{ j * m_fDx - m_fHalfWidth,A,-i * m_fDz + m_fHalfWidth,1.f };
        vector_t p1 = vector_t{ (j + 1) * m_fDx - m_fHalfWidth,B,-i * m_fDz + m_fHalfWidth,1.f };
        vector_t p2 = vector_t{ j * m_fDx - m_fHalfWidth,C,-(i + 1) * m_fDz + m_fHalfWidth,1.f };

        vector_t vPlane = XMPlaneFromPoints(p0, p1, p2);
        float4_t vReturnPlane;
        XMStoreFloat4(&vReturnPlane, vPlane);
        return vReturnPlane;
    }
    else
    {
        vector_t p0 = vector_t{ j * m_fDx - m_fHalfWidth,C,-(i + 1) * m_fDz + m_fHalfWidth,1.f };
        vector_t p1 = vector_t{ (j + 1) * m_fDx - m_fHalfWidth,B,-i * m_fDz + m_fHalfWidth,1.f };
        vector_t p2 = vector_t{ (j + 1) * m_fDx - m_fHalfWidth,D,-(i + 1) * m_fDz + m_fHalfWidth,1.f };

        vector_t vPlane = XMPlaneFromPoints(p0, p1, p2);
        float4_t vReturnPlane;
        XMStoreFloat4(&vReturnPlane, vPlane);
        return vReturnPlane;
    }
}
```

> float3_t CHill::GetNormal(uint32_t i, uint32_t j)

- 현재 정점위치의 위,아래 / 왼쪽,오른쪽의 정점을 이용하여 법선벡터를 구한다.
- 만약 위,아래/ 왼쪽, 오른쪽에 정점이 없는 테두리에 있는 정점이라면 자신의 위치를 사용한다.
- 각각의 차분을 이용하여 법선 벡터를 구한다.

```cpp
float3_t CHill::GetNormal(uint32_t i, uint32_t j)
{
    // 중심 차분
    // i -> 행 , j -> 열
    uint32_t iUp = (i > 0) ? i - 1 : i; // i<=0 이면 up은 자기 자신이 되는 것
    uint32_t iDown = (i < m_ivtxCntZ - 1) ? i + 1 : i;
    uint32_t jLeft = (j > 0) ? j - 1 : j;
    uint32_t jRight = (j < m_ivtxCntX - 1) ? j + 1 : j;

    f32_t L = m_vHeightmap[i * m_ivtxCntX + jLeft];
    f32_t R = m_vHeightmap[i * m_ivtxCntX + jRight];
    f32_t U = m_vHeightmap[iUp * m_ivtxCntX + j];      // +z 쪽
    f32_t D = m_vHeightmap[iDown * m_ivtxCntX + j];    // -z 쪽

    // 실제 이웃 간 거리 (경계에서는 한 칸, 내부에서는 두 칸)
    f32_t fDistX = (jRight - jLeft) * m_fDx;
    f32_t fDistZ = (iDown - iUp) * m_fDz;

    vector_t vN = vector_t{ (L - R) / fDistX, 
                            1.f, 
                            (D - U) / fDistZ, 
                            0.f };

    float3_t vNormal;
    XMStoreFloat3(&vNormal, XMVector3Normalize(vN));
    return vNormal;
}
```

  <summary> 3. 혼합맵 </summary>

> 혼합맵이란?

- 여러장의 텍스처(계층 텍스처)를 섞어서 만들 수 있는 텍스처의 혼합 기준이 되는 텍스처.
- 계층 텍스처는 타일 방식으로 적용하게 된다.

> 적용 - 1. 텍스처 준비

- 계층 텍스처와 혼합맵 텍스처를 준비한다.
- 계층 텍스처는 벡터로 저장한 뒤 셰이더에 하나씩 전달한다.
```cpp
if (FAILED(__super::Add_Component(ETOUI(LEVEL::GAMEPLAY), TEXT("Prototype_Component_Texture_Terrain_Blend"),
    TEXT("Com_Texture"), m_pBlendCom)))
    return E_FAIL;

if (FAILED(__super::Add_Component(ETOUI(LEVEL::GAMEPLAY), TEXT("Prototype_Component_Texture_Terrain_Layer"),
    TEXT("Com_Blend"), m_pLayerCom)))
    return E_FAIL;
```

> 적용 - 2. 자원 바인딩

- 계층 텍스처는 하나씩 슬롯에 넣어주고, 혼합맵 텍스처는 한 번에 넣어준다.
```cpp
// 계층 텍스처
m_pLayerCom->Set_TexturePS(0, 2);    // Layer0 잔디     → t2
m_pLayerCom->Set_TexturePS(1, 3);    // Layer1 흙       → t3
m_pLayerCom->Set_TexturePS(2, 4);    // Layer2 아스팔트 → t4

// 혼합맵 텍스처
m_pBlendCom->Set_TexturePS(0, 5);    // 혼합맵          → t5
```

> 적용 - 3. UV값 설정

- 혼합맵은 맵 전체에 한 장 덮어야 하므로 (0,0) ~ (1,1)로 UV값을 설정한다.
```cpp
m_tMeshData.Vertices[i * m_ivtxCntX + j].TexC.x = j*du;
m_tMeshData.Vertices[i * m_ivtxCntX + j].TexC.y = i*dv;
```

> 적용 - 4. 셰이더 파일
- 텍스처를 받을 슬롯을 설정해주고 계층 텍스처를 몇 장 깔지 반복 횟수를 설정해준다

```hlsl
// 계층 텍스처 
Texture2D g_LayerMap0 : register(t2); // 잔디     (blend.r)
Texture2D g_LayerMap1 : register(t3); // 흙       (blend.g)
Texture2D g_LayerMap2 : register(t4); // 아스팔트  (blend.b)

// 혼합맵 텍스처
Texture2D g_BlendMap : register(t5);

// 지형 전체에 레이어를 반복할 횟수 (지형 크기 : 320(정점 수) * 2(간격) / 잔디 크기 tileSize 4m)
static const float2 g_vTexScale = float2(160.f, 160.f); // 가로 세로 160장씩 깔 예정
```
- 혼합맵의 rgb값을 읽어야하고, 계층 텍스처용 UV값이 필요하므로 VS_OUT_TERRAIN 구조체를 추가한다.
```cpp
struct VS_OUT_TERRAIN
{
    float4 vPosition : SV_POSITION;
    float3 vPosW : POSITION;
    float3 vNormalW : NORMAL;
    float2 vTex : TEXCOORD0; // 혼합맵용 0~1    
    float2 vTiledTex : TEXCOORD1; // 레이어용 UV (0 ~ 160)
};
```

- VS_TERRAIN 버텍스 셰이더 진입함수를 만들어서 VS_OUT_TERRAIN에 만든 계층 텍스처의 UV값을 채워준다
```cpp
VS_OUT_TERRAIN VS_TERRAIN(VS_IN In)
{
    VS_OUT_TERRAIN Out;

    float4 vPosW = mul(float4(In.vPosition, 1.f), g_matWorld);
    Out.vPosW = vPosW.xyz;
    Out.vNormalW = mul(In.vNormal, (float3x3) g_matWorldInvTranspose);
    Out.vPosition = mul(mul(vPosW, g_matView), g_matProj);

    Out.vTex = In.vTexCoord;
    Out.vTiledTex = In.vTexCoord * g_vTexScale; // 계층 텍스처의 UV값을 설정한다.
    return Out;
}
```

- PS_TERRAIN 픽셀 셰이더 진입함수에서는 혼합맵의 r,g,b값의 비율대로 계층 텍스처를 섞어서 픽셀 색을 정한다.
```cpp
// Sample layers.
float4 c0 = g_LayerMap0.Sample(g_Sampler, In.vTiledTex);    // 잔디
float4 c1 = g_LayerMap1.Sample(g_Sampler, In.vTiledTex);    // 흙
float4 c2 = g_LayerMap2.Sample(g_Sampler, In.vTiledTex);    // 아스팔트

// Sample the blend map.
float4 t = g_BlendMap.Sample(g_Sampler, In.vTex);

// 잔디 x R + 흙 x G + 아스팔트 x B
float4 vTexColor = c0 * t.r + c1 * t.g + c2 * t.b;
```
